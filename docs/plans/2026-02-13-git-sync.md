# Git Sync Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Add opt-in git tracking to inkan's data directory (`~/.config/inkan/`) with auto-commit on save, pull on startup, and push on exit.

**Architecture:** A new `src/git.ts` module handles all git operations via `execSync`. The existing `storage.ts` save functions call `gitCommit()` after writes. `App.tsx` calls `gitPull()` on mount and `gitPush()` on exit. The `/git` command family is added to `commands.ts` and `CommandBar.tsx`.

**Tech Stack:** Node.js `child_process.execSync`, no new dependencies.

### Task 1: Create `src/git.ts` — core git operations

**Files:**
- Create: `src/git.ts`

**Step 1: Create the git utility module**

```typescript
import { execSync } from "node:child_process";
import fs from "node:fs";
import path from "node:path";
import os from "node:os";

const CONFIG_DIR = path.join(os.homedir(), ".config", "inkan");

export function isGitRepo(): boolean {
  return fs.existsSync(path.join(CONFIG_DIR, ".git"));
}

export function hasRemote(): boolean {
  if (!isGitRepo()) return false;
  try {
    const result = execSync("git remote get-url origin", { cwd: CONFIG_DIR, stdio: "pipe" }).toString().trim();
    return result.length > 0;
  } catch {
    return false;
  }
}

export function gitInit(): { success: boolean; message: string } {
  try {
    fs.mkdirSync(CONFIG_DIR, { recursive: true });
    execSync("git init", { cwd: CONFIG_DIR, stdio: "pipe" });
    execSync("git add -A", { cwd: CONFIG_DIR, stdio: "pipe" });
    execSync('git commit -m "Initial commit"', { cwd: CONFIG_DIR, stdio: "pipe" });
    return { success: true, message: "Git initialized in data directory" };
  } catch (e: unknown) {
    const msg = e instanceof Error ? e.message : String(e);
    return { success: false, message: `Git init failed: ${msg}` };
  }
}

export function gitSetRemote(url: string): { success: boolean; message: string } {
  if (!isGitRepo()) return { success: false, message: "Not a git repo — run /git init first" };
  try {
    if (hasRemote()) {
      execSync(`git remote set-url origin ${url}`, { cwd: CONFIG_DIR, stdio: "pipe" });
    } else {
      execSync(`git remote add origin ${url}`, { cwd: CONFIG_DIR, stdio: "pipe" });
    }
    return { success: true, message: `Remote set to ${url}` };
  } catch (e: unknown) {
    const msg = e instanceof Error ? e.message : String(e);
    return { success: false, message: `Failed to set remote: ${msg}` };
  }
}

export function gitCommit(message: string): { success: boolean; message: string } {
  if (!isGitRepo()) return { success: true, message: "" }; // silently skip
  try {
    execSync("git add -A", { cwd: CONFIG_DIR, stdio: "pipe" });
    // Check if there's anything to commit
    try {
      execSync("git diff --cached --quiet", { cwd: CONFIG_DIR, stdio: "pipe" });
      return { success: true, message: "" }; // nothing to commit
    } catch {
      // diff --quiet exits non-zero when there are staged changes
    }
    execSync(`git commit -m "${message.replace(/"/g, '\\"')}"`, { cwd: CONFIG_DIR, stdio: "pipe" });
    return { success: true, message: "" };
  } catch (e: unknown) {
    const msg = e instanceof Error ? e.message : String(e);
    return { success: false, message: `Git commit failed: ${msg}` };
  }
}

export function gitPull(): { success: boolean; message: string } {
  if (!isGitRepo() || !hasRemote()) return { success: true, message: "" };
  try {
    execSync("git pull --rebase", { cwd: CONFIG_DIR, stdio: "pipe" });
    return { success: true, message: "" };
  } catch (e: unknown) {
    const msg = e instanceof Error ? e.message : String(e);
    return { success: false, message: `Git pull failed: ${msg}` };
  }
}

export function gitPush(): { success: boolean; message: string } {
  if (!isGitRepo() || !hasRemote()) return { success: true, message: "" };
  try {
    execSync("git push", { cwd: CONFIG_DIR, stdio: "pipe" });
    return { success: true, message: "" };
  } catch (e: unknown) {
    const msg = e instanceof Error ? e.message : String(e);
    return { success: false, message: `Git push failed: ${msg}` };
  }
}

export function gitStatus(): { initialized: boolean; remote: string | null; clean: boolean } {
  if (!isGitRepo()) return { initialized: false, remote: null, clean: true };
  let remote: string | null = null;
  try {
    remote = execSync("git remote get-url origin", { cwd: CONFIG_DIR, stdio: "pipe" }).toString().trim();
  } catch { /* no remote */ }
  let clean = true;
  try {
    execSync("git diff --quiet && git diff --cached --quiet", { cwd: CONFIG_DIR, stdio: "pipe", shell: "/bin/bash" });
  } catch {
    clean = false;
  }
  return { initialized: true, remote, clean };
}
```

### Task 4: Add `/git` commands to `commands.ts`

**Files:**

**Step 1: Extend CommandContext and executeCommand**

Add to `CommandContext` interface:
```typescript

```

Add `case "git":` to the switch in `executeCommand`:
```typescript
case "git": {
  const sub = parts[1]?.toLowerCase();
  if (!sub) return { success: false, message: "Usage: /git [init|remote|push|pull|status]" };

  if (sub === "init") {
    return ctx.gitInit();
  }
  if (sub === "remote") {
    const url = parts.slice(2).join(" ");
    if (!url) return { success: false, message: "Usage: /git remote <url>" };
    return ctx.gitSetRemote(url);
  }
  if (sub === "push") {
    return ctx.gitPush();
  }
  if (sub === "pull") {
    return ctx.gitPull();
  }
  if (sub === "status") {
    const s = ctx.gitStatus();
    if (!s.initialized) return { success: true, message: "Git: not initialized — run /git init" };
    const remote = s.remote ?? "none";
    const status = s.clean ? "clean" : "uncommitted changes";
    return { success: true, message: `Git: remote=${remote}, ${status}` };
  }
  return { success: false, message: `Unknown git subcommand: ${sub}` };
}
```
